# Omnisearch v1.1 — New Providers & Parallel Search Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add four search providers (Tavily, Google Programmable Search, SearXNG, Perplexity) and opt-in parallel multi-provider search with merged results.

**Architecture:** Each new provider is one file following the v1 `SearchProvider` + DI pattern, wired into the existing lazy registry. Parallel search lives inside the router (`search(query, options?, chain?, parallel?)`): `Promise.allSettled` over participant providers, per-participant cache entries and rate limiters, round-robin merge with URL dedupe, composite merged-cache key. All three surfaces (MCP tool, CLI, plugins) pass the new `parallel` boolean through to the router.

**Tech Stack:** Bun + TypeScript (NodeNext), zod 4.1.8 (pinned via root override), `@modelcontextprotocol/sdk`, OpenAI SDK for Perplexity.

**Spec:** `docs/superpowers/specs/2026-09-02-v1.1-providers-parallel-search-design.md`

## Global Constraints

- `ProviderName` union is exactly: `"anthropic" | "openai" | "brave" | "moonshot" | "tavily" | "google" | "searxng" | "perplexity"`.
- Every provider: structural DI for tests (`fetchImpl` / `client` / `importSdk` injection), fail-fast credential validation at construction, HTML stripped from snippets (`stripHtml` from `../util.js`), errors of the form `<Provider> search failed: HTTP <status> — <first 200 chars>`.
- Tests mock the network — no live API calls anywhere.
- zod stays pinned at `4.1.8` via the root `package.json` `overrides` — do not touch dependency versions.
- No new npm dependencies.
- Repo commands run from repo root with Git Bash: `bun test`, `bun run typecheck`, `bun run build`. Relative TS imports use `.js` extensions (NodeNext).
- Single-provider behavior and all existing v1 tests must remain green untouched (only the `USAGE` string in `cli.ts` changes, which its tests import as a constant).
- Final package versions: `0.2.0` (root + all four packages), set in Task 9.

```typescript
import type { SearchOptions, SearchProvider, SearchResult } from "../types.js";
import { resolveApiKey, stripHtml } from "../util.js";

const TAVILY_ENDPOINT = "https://api.tavily.com/search";
const DEFAULT_MAX_RESULTS = 10;
const MAX_RESULTS = 20;

interface TavilyApiResponse {
  results?: Array<{ title?: string; url?: string; content?: string }>;
}

export interface TavilyProviderOptions {
  /** Tavily API key. Defaults to the TAVILY_API_KEY environment variable. */
  apiKey?: string;
  /** Injectable fetch for tests. */
  fetchImpl?: typeof fetch;
}

/**
 * Tavily provider backed by the pure REST API — no SDK required.
 */
export function createTavilyProvider(options: TavilyProviderOptions = {}): SearchProvider {
  const fetchImpl = options.fetchImpl ?? fetch;
  const apiKey = resolveApiKey("tavily", options.apiKey);

  return {
    name: "tavily",

    async search(query: string, searchOptions: SearchOptions = {}): Promise<SearchResult[]> {
      const maxResults = Math.min(
        Math.max(1, searchOptions.maxResults ?? DEFAULT_MAX_RESULTS),
        MAX_RESULTS,
      );

      const response = await fetchImpl(TAVILY_ENDPOINT, {
        method: "POST",
        headers: {
          Authorization: `Bearer ${apiKey}`,
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ query, max_results: maxResults }),
      });

      if (!response.ok) {
        const detail = (await response.text()).slice(0, 200);
        throw new Error(
          `Tavily search failed: HTTP ${response.status}${detail ? ` — ${detail}` : ""}`,
        );
      }

      const data = (await response.json()) as TavilyApiResponse;
      return (data.results ?? []).map((result) => ({
        title: result.title ?? "",
        url: result.url ?? "",
        snippet: stripHtml(result.content ?? ""),
      }));
    },
  };
}
```

```typescript
import type { SearchOptions, SearchProvider, SearchResult } from "../types.js";
import { requireEnv, resolveApiKey, stripHtml } from "../util.js";

const GOOGLE_ENDPOINT = "https://www.googleapis.com/customsearch/v1";
const DEFAULT_MAX_RESULTS = 10;
const GOOGLE_MAX_RESULTS = 10;

interface GoogleApiResponse {
  items?: Array<{ title?: string; link?: string; snippet?: string }>;
}

export interface GoogleProviderOptions {
  /** Google API key. Defaults to the GOOGLE_API_KEY environment variable. */
  apiKey?: string;
  /** Programmable Search Engine ID. Defaults to GOOGLE_SEARCH_ENGINE_ID. */
  searchEngineId?: string;
  /** Injectable fetch for tests. */
  fetchImpl?: typeof fetch;
}

/**
 * Google Programmable Search provider. Needs both an API key and a search
 * engine ID (cx); the registry validates the key, this factory validates the cx.
 */
export function createGoogleProvider(options: GoogleProviderOptions = {}): SearchProvider {
  const fetchImpl = options.fetchImpl ?? fetch;
  const apiKey = resolveApiKey("google", options.apiKey);
  const searchEngineId = options.searchEngineId ?? requireEnv("GOOGLE_SEARCH_ENGINE_ID");

  return {
    name: "google",

    async search(query: string, searchOptions: SearchOptions = {}): Promise<SearchResult[]> {
      const num = Math.min(
        Math.max(1, searchOptions.maxResults ?? DEFAULT_MAX_RESULTS),
        GOOGLE_MAX_RESULTS,
      );
      const params = new URLSearchParams({
        key: apiKey,
        cx: searchEngineId,
        q: query,
        num: String(num),
      });

      const response = await fetchImpl(`${GOOGLE_ENDPOINT}?${params}`, {
        headers: { Accept: "application/json" },
      });

      if (!response.ok) {
        const detail = (await response.text()).slice(0, 200);
        throw new Error(
          `Google search failed: HTTP ${response.status}${detail ? ` — ${detail}` : ""}`,
        );
      }

      const data = (await response.json()) as GoogleApiResponse;
      return (data.items ?? []).map((item) => ({
        title: item.title ?? "",
        url: item.link ?? "",
        snippet: stripHtml(item.snippet ?? ""),
      }));
    },
  };
}
```

```typescript
import type { SearchOptions, SearchProvider, SearchResult } from "../types.js";
import { requireEnv, stripHtml } from "../util.js";

const DEFAULT_MAX_RESULTS = 10;

interface SearxngApiResponse {
  results?: Array<{ title?: string; url?: string; content?: string }>;
}

export interface SearxngProviderOptions {
  /** SearXNG instance base URL. Defaults to SEARXNG_BASE_URL. */
  baseUrl?: string;
  /** Injectable fetch for tests. */
  fetchImpl?: typeof fetch;
}

/**
 * SearXNG provider for self-hosted instances. The instance must have the
 * JSON format enabled (settings: formats: ["html", "json"]).
 */
export function createSearxngProvider(options: SearxngProviderOptions = {}): SearchProvider {
  const fetchImpl = options.fetchImpl ?? fetch;
  const baseUrl = (options.baseUrl ?? requireEnv("SEARXNG_BASE_URL")).replace(/\/+$/, "");

  return {
    name: "searxng",

    async search(query: string, searchOptions: SearchOptions = {}): Promise<SearchResult[]> {
      const params = new URLSearchParams({ q: query, format: "json" });

      const response = await fetchImpl(`${baseUrl}/search?${params}`, {
        headers: { Accept: "application/json" },
      });

      if (!response.ok) {
        const detail = (await response.text()).slice(0, 200);
        const hint =
          response.status === 403
            ? " (the instance must enable the JSON format: settings -> formats: [\"html\", \"json\"])"
            : "";
        throw new Error(
          `SearXNG search failed: HTTP ${response.status}${hint}${detail ? ` — ${detail}` : ""}`,
        );
      }

      const data = (await response.json()) as SearxngApiResponse;
      const mapped = (data.results ?? []).map((result) => ({
        title: result.title ?? "",
        url: result.url ?? "",
        snippet: stripHtml(result.content ?? ""),
      }));
      return mapped.slice(0, searchOptions.maxResults ?? DEFAULT_MAX_RESULTS);
    },
  };
}
```

```typescript
import type { SearchOptions, SearchProvider, SearchResult } from "../types.js";
import { loadOptionalSdk, resolveApiKey } from "../util.js";

const PERPLEXITY_BASE_URL = "https://api.perplexity.ai";
const PERPLEXITY_MODEL = "sonar";

export const OPENAI_SDK_INSTALL_HINT =
  "OpenAI SDK not installed. Run: npm install openai (required for the Perplexity provider)";

export interface PerplexityMessage {
  content?: string | null;
  search_results?: Array<{ title?: string; url?: string }>;
  annotations?: Array<{ url_citation?: { title?: string; url?: string } }>;
  citations?: unknown;
}

export interface PerplexityChatResponse {
  choices?: Array<{ message?: PerplexityMessage }>;
}

/** Minimal structural view of the OpenAI-compatible Perplexity client. */
export interface PerplexityClientLike {
  chat: {
    completions: {
      create(params: Record<string, unknown>): Promise<PerplexityChatResponse>;
    };
  };
}

export interface PerplexityProviderOptions {
  /** Perplexity API key. Defaults to the PERPLEXITY_API_KEY environment variable. */
  apiKey?: string;
  /** Pre-built client; used directly when provided (tests). */
  client?: PerplexityClientLike;
  /** Injectable SDK loader so a missing install can be simulated in tests. */
  importSdk?: () => Promise<PerplexitySdkModule>;
}

interface PerplexitySdkModule {
  default: new (options: { apiKey: string; baseURL: string }) => PerplexityClientLike;
}

async function dynamicImportSdk(): Promise<PerplexitySdkModule> {
  return (await import("openai")) as unknown as PerplexitySdkModule;
}

function loadSdk(importSdk?: PerplexityProviderOptions["importSdk"]): Promise<PerplexitySdkModule> {
  return loadOptionalSdk(importSdk ?? dynamicImportSdk, OPENAI_SDK_INSTALL_HINT);
}

/** Citations priority: search_results → annotations[].url_citation → bare citation URLs. */
function extractResults(message: PerplexityMessage | undefined, maxResults?: number): SearchResult[] {
  const results: SearchResult[] = [];
  for (const item of message?.search_results ?? []) {
    if (!item?.url) continue;
    results.push({ title: item.title ?? item.url, url: item.url, snippet: "" });
  }
  if (results.length === 0) {
    for (const annotation of message?.annotations ?? []) {
      const citation = annotation?.url_citation;
      if (!citation?.url) continue;
      results.push({ title: citation.title ?? citation.url, url: citation.url, snippet: "" });
    }
  }
  if (results.length === 0 && Array.isArray(message?.citations)) {
    for (const citation of message.citations) {
      if (typeof citation === "string") {
        results.push({ title: citation, url: citation, snippet: "" });
      }
    }
  }
  return maxResults ? results.slice(0, maxResults) : results;
}

/**
 * Perplexity provider: OpenAI-compatible endpoint that searches automatically
 * (no tool loop). LLM-backed — costed like the other LLM providers.
 */
export async function createPerplexityProvider(
  options: PerplexityProviderOptions = {},
): Promise<SearchProvider> {
  const apiKey = resolveApiKey("perplexity", options.apiKey);
  const client =
    options.client ??
    new (await loadSdk(options.importSdk)).default({ apiKey, baseURL: PERPLEXITY_BASE_URL });

  return {
    name: "perplexity",

    async search(query: string, searchOptions: SearchOptions = {}): Promise<SearchResult[]> {
      const response = await client.chat.completions.create({
        model: PERPLEXITY_MODEL,
        messages: [{ role: "user", content: query }],
      });
      return extractResults(response.choices?.[0]?.message, searchOptions.maxResults);
    },
  };
}
```

```typescript
export function mergedCacheKey(
```

```typescript
  function resolveParticipants(): ProviderName[] {
    const explicit = config.parallelParticipants;
    if (explicit && explicit.length > 0) return explicit;
    const scanned = AVAILABLE_PROVIDERS.filter(hasProviderConfig);
    return scanned;
  }

  function interleave(grouped: Array<{ name: ProviderName; results: SearchResult[] }>, cap: number): SearchResult[] {
    const merged: SearchResult[] = [];
    const seen = new Set<string>();
    const depth = Math.max(0, ...grouped.map((g) => g.results.length));
    for (let rank = 0; rank < depth && merged.length < cap; rank++) {
      for (const group of grouped) {
        if (merged.length >= cap) break;
        const result = group.results[rank];
        if (!result || seen.has(result.url)) continue;
        seen.add(result.url);
        merged.push(result);
      }
    }
    return merged;
  }

  async function searchParallel(query: string, options: SearchOptions): Promise<SearchResponse> {
    const participants = resolveParticipants();
    if (participants.length === 0) {
      return search(query, options);
    }

    const key = mergedCacheKey(participants, query, options);
    const cached = cache.get(key);
    if (cached) {
      return {
        query,
        provider: cached.provider,
        requestedProvider: cached.requestedProvider,
        fellBack: cached.fellBack,
        results: cached.results,
        markdown: cached.markdown,
        cached: true,
        elapsedMs: 0,
        mode: "parallel",
        providers: cached.providers,
        failedProviders: cached.failedProviders,
      };
    }

    const startedAt = Date.now();
    const tasks = participants.map(async (name) => {
      const perKey = cacheKey(name, query, options);
      const perCached = cache.get(perKey);
      if (perCached) return perCached.results;
      const provider = await providerFor(name);
      await limiterFor(name)?.acquire();
      const results = await provider.search(query, options);
      cache.set(perKey, {
        provider: name,
        requestedProvider: undefined,
        fellBack: false,
        results,
        markdown: normalizeResults(query, name, results),
      });
      return results;
    });
    const settled = await Promise.allSettled(tasks);

    const succeeded: Array<{ name: ProviderName; results: SearchResult[] }> = [];
    const failed: ProviderName[] = [];
    settled.forEach((outcome, index) => {
      const name = participants[index] as ProviderName;
      if (outcome.status === "fulfilled") {
        succeeded.push({ name, results: outcome.value });
      } else {
        failed.push(name);
      }
    });

    if (succeeded.length === 0) {
      const errors = settled.map((outcome, index) => {
        const name = participants[index] as ProviderName;
        const message =
          outcome.status === "rejected"
            ? outcome.reason instanceof Error
              ? outcome.reason.message
              : String(outcome.reason)
            : "unexpected fulfillment";
        return `${name}: ${message}`;
      });
      throw new Error(`All search providers failed — ${errors.join(" | ")}`);
    }

    const merged = interleave(succeeded, options.maxResults ?? 10);
    const succeededNames = succeeded.map((entry) => entry.name);
    let markdown = normalizeResults(query, succeededNames[0] as ProviderName, merged, `parallel: ${succeededNames.join(", ")}`);
    markdown = appendFailureNote(markdown, failed);

    const value: CacheValue = {
      provider: participants[0] as ProviderName,
      requestedProvider: undefined,
      fellBack: false,
      results: merged,
      markdown,
      mode: "parallel",
      providers: participants,
      failedProviders: failed,
    };
    cache.set(key, value);

    return {
      query,
      provider: value.provider,
      requestedProvider: value.requestedProvider,
      fellBack: value.fellBack,
      results: merged,
      markdown,
      cached: false,
      elapsedMs: Date.now() - startedAt,
      mode: "parallel",
      providers: participants,
      failedProviders: failed,
    };
  }
```
